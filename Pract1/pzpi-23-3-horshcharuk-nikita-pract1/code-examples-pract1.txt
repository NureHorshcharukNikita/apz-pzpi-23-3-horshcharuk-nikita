Реалізація містить інтерфейс стратегії, конкретні класи стратегій для додавання, віднімання та множення, а також клас контексту для делегування виконання алгоритму. 1  using System;
2  
3  namespace StrategyPatternExample
4  {
5      public interface IStrategy
6      {
7          int Execute(int a, int b);
8      }
9  
10     public class AdditionStrategy : IStrategy
11     {
12         public int Execute(int a, int b)
13         {
14             return a + b;
15         }
16     }
17 
18     public class SubtractionStrategy : IStrategy
19     {
20         public int Execute(int a, int b)
21         {
22             return a - b;
23         }
24     }
25 
26     public class MultiplicationStrategy : IStrategy
27     {
28         public int Execute(int a, int b)
29         {
30             return a * b;
31         }
32     }
33 
34     public class Context
35     {
36         private IStrategy _strategy;
37 
38         public Context(IStrategy strategy)
39         {
40             _strategy = strategy;
41         }
42 
43         public void SetStrategy(IStrategy strategy)
44         {
45             _strategy = strategy;
46         }
47 
48         public int ExecuteStrategy(int a, int b)
49         {
50             return _strategy.Execute(a, b);
51         }
52     }
53 
54     internal class Program
55     {
56         static void Main(string[] args)
57         {
58             Context context = new Context(new AdditionStrategy());
59             Console.WriteLine("Додавання: " + context.ExecuteStrategy(10, 5));
60 
61             context.SetStrategy(new SubtractionStrategy());
62             Console.WriteLine("Віднімання: " + context.ExecuteStrategy(10, 5));
63 
64             context.SetStrategy(new MultiplicationStrategy());
65             Console.WriteLine("Множення: " + context.ExecuteStrategy(10, 5));
66         }
